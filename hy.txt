  <script>

    document.addEventListener("DOMContentLoaded", function () {

      /* =========================
         MOBILE MENU
      ========================= */

      const menuIcon = document.getElementById("menuIcon");
      const nav = document.getElementById("nav");

      menuIcon.addEventListener("click", function (event) {

        event.stopPropagation();

        nav.classList.toggle("active");

        /*
          Close all dropdowns
          when mobile menu is toggled.
        */

        closeAllDropdowns();

      });


      /* =========================
         CLICK DROPDOWNS
      ========================= */

      const clickDropdowns =
        document.querySelectorAll(".click-dropdown");


      clickDropdowns.forEach(function (dropdown) {

        const button =
          dropdown.querySelector(".dropdown-toggle-btn");

        const menu =
          dropdown.querySelector(".dropdown-menu");


        button.addEventListener("click", function (event) {

          event.stopPropagation();


          /*
            Close other click dropdowns
          */

          clickDropdowns.forEach(function (otherDropdown) {

            if (otherDropdown !== dropdown) {

              const otherMenu =
                otherDropdown.querySelector(".dropdown-menu");

              otherMenu.classList.remove("show");

            }

          });


          /*
            Toggle current dropdown
          */

          menu.classList.toggle("show");

        });

      });


      /* =========================
         CLOSE WHEN CLICKING OUTSIDE
      ========================= */

      document.addEventListener("click", function (event) {

        /*
          If click is outside any dropdown,
          close click dropdowns.
        */

        if (!event.target.closest(".dropdown")) {

          closeAllDropdowns();

        }

      });


      /* =========================
         CLOSE DROPDOWNS FUNCTION
      ========================= */

      function closeAllDropdowns() {

        document
          .querySelectorAll(".click-dropdown .dropdown-menu")
          .forEach(function (menu) {

            menu.classList.remove("show");

          });

      }


      /* =========================
         MOBILE NAV LINK CLICK
      ========================= */

      document
        .querySelectorAll(".center-nav .dropdown-btn")
        .forEach(function (button) {

          button.addEventListener("click", function () {

            /*
              On mobile, hover dropdowns
              should also work with click.
            */

            if (window.innerWidth <= 780) {

              const parent =
                button.closest(".dropdown");

              const menu =
                parent.querySelector(".dropdown-menu");


              /*
                Don't interfere with
                click dropdowns.
              */

              if (!parent.classList.contains("click-dropdown")) {

                /*
                  Close other menus
                */

                document
                  .querySelectorAll(".dropdown-menu")
                  .forEach(function (otherMenu) {

                    if (otherMenu !== menu) {
                      otherMenu.classList.remove("show");
                    }

                  });


                menu.classList.toggle("show");

              }

            }

          });

        });


      /* =========================
         WINDOW RESIZE
      ========================= */

      window.addEventListener("resize", function () {

        if (window.innerWidth > 780) {

          nav.classList.remove("active");

        }

      });

    });

  </script>
